SQL生成语句 · 可编辑执行复制
-- 语义模型: 客单价 = GMV / 订单数 (metric_formulas)
SELECT cate_level1_name,
       SUM(gmv)/SUM(order_cnt) AS avg_order_value
FROM ads_ctlive_item_stats
WHERE stat_period = '30d'
GROUP BY cate_level1_name
ORDER BY avg_order_value DESC;
